# Python packages the faiss Python wheel build and its tests need, installed into the toolkit's build venv
# (tuned/build/<cuda-tag>/venv) by tuned/wheel.sh. Before this file they came from whatever the system python3 held.
# The versions are locked in requirements-build.lock (gpu_tuned_freeze_requirements) and installed under it; to change them
# build with GPU_TUNED_UNLOCKED=1 and freeze again.

# prerequisite message asked for conda's swig=4.0, so this is a deliberate match to what actually built, not to the message.
swig==4.2.0

# faiss's setup.py and tests import numpy; two test files import scipy.
numpy
scipy

# full_test.sh and smoke_test.sh run pytest.
pytest

# setup.py bdist_wheel.
wheel

# The cuVS and RMM CMake config files in the packages we consume need CMake 4.0; the host has 3.30.
cmake>=4.0
